26/// Writer is designed to write data into given path in an asynchronous
27/// manner.
28///
29/// ## Notes
30///
31/// Please make sure either `close` or `abort` has been called before
32/// dropping the writer otherwise the data could be lost.
33///
34/// ## Usage
35///
36/// ### Write Multiple Chunks
37///
38/// Sending multiple chunks to the service requires [`Capability::write_can_multi`].
39/// A writer that accepts only one chunk returns [`ErrorKind::Unsupported`] if it
40/// receives a second chunk. See [`Operator::writer_with`] for buffering options.
41///
42/// ```
43/// use opendal_core::Operator;
44/// use opendal_core::Result;
45///
46/// async fn test(op: Operator) -> Result<()> {
47///     let mut w = op.writer("path/to/file").await?;
48///     w.write(vec![1; 1024]).await?;
49///     w.write(vec![2; 1024]).await?;
50///     w.close().await?;
51///     Ok(())
52/// }
53/// ```
54///
55/// ### Write like `Sink`
56///
57/// ```
58/// use anyhow::Result;
59/// use futures::SinkExt;
60/// use opendal_core::Operator;
61///
62/// async fn test(op: Operator) -> Result<()> {
63///     let mut w = op.writer("path/to/file").await?.into_bytes_sink();
64///     w.send(vec![1; 1024].into()).await?;
65///     w.send(vec![2; 1024].into()).await?;
66///     w.close().await?;
67///     Ok(())
68/// }
69/// ```
70///
71/// ### Write like `AsyncWrite`
72///
73/// ```
74/// use anyhow::Result;
75/// use futures::AsyncWriteExt;
76/// use opendal_core::Operator;
77///
78/// async fn test(op: Operator) -> Result<()> {
79///     let mut w = op.writer("path/to/file").await?.into_futures_async_write();
80///     w.write(&vec![1; 1024]).await?;
81///     w.write(&vec![2; 1024]).await?;
82///     w.close().await?;
83///     Ok(())
84/// }
85/// ```
86///
87/// ### Write with append enabled
88///
89/// Writer also supports to write with append enabled. This is useful when users want to append
90/// some data to the end of the file.
91///
92/// - If file doesn't exist, it will be created and just like calling `write`.
93/// - If file exists, data will be appended to the end of the file.
94///
95/// Possible Errors:
96///
97/// - Some services store normal file and appendable file in different way. Trying to append
98///   on non-appendable file could return [`ErrorKind::Conflict`] error.
99/// - Services that doesn't support append will return [`ErrorKind::Unsupported`] error when
100///   creating writer with `append` enabled.
101pub struct Writer {
102    /// Keep a reference to write context in writer.
103    ctx: Arc<WriteContext>,
104    inner: WriteState,
105}

110            .field("chunk", &self.ctx.options().chunk())
111            .field("append", &self.ctx.args().append())
112            .finish_non_exhaustive()
113    }
114}

125    /// Write [`Buffer`] into writer.
126    ///
127    /// This operation will write all data in given buffer into writer.
128    ///
129    /// When a single call spans multiple chunks, a failure after the underlying
130    /// writer accepts part of the input returns a persistent error. Call
131    /// [`Writer::abort`] and start a new writer if you need to retry that write.
132    ///
133    /// ## Examples
134    ///
135    /// ```
136    /// use bytes::Bytes;
137    /// use opendal_core::Operator;
138    /// use opendal_core::Result;
139    ///
140    /// async fn test(op: Operator) -> Result<()> {
141    ///     let mut w = op.writer("hello.txt").await?;
142    ///     // Buffer can be created from continues bytes.
143    ///     w.write("hello, world").await?;
144    ///     // Buffer can also be created from non-continues bytes.
145    ///     w.write(vec![Bytes::from("hello,"), Bytes::from("world!")])
146    ///         .await?;
147    ///
148    ///     // Make sure file has been written completely.
149    ///     w.close().await?;
150    ///     Ok(())
151    /// }
152    /// ```
153    pub async fn write(&mut self, bs: impl Into<Buffer>) -> Result<()> {
154        self.inner.write_all(bs.into()).await
155    }

157    /// Write [`bytes::Buf`] into inner writer.
158    ///
159    /// This operation will write all data in given buffer into writer.
160    ///
161    /// # Notes
162    ///
163    /// This function iterates over each chunk in the `Buf` and collects them
164    /// into a `Buffer`. For `Bytes` chunks, `copy_to_bytes` is zero-copy
165    /// (only a ref-count increment). For other chunk types (e.g., `&[u8]`),
166    /// the data will be copied.
167    ///
168    /// This approach minimizes copies for chained buffers (`Chain<Bytes, Bytes>`)
169    /// where each individual chunk can be extracted without copying.
170    pub async fn write_from(&mut self, bs: impl Buf) -> Result<()> {
171        let mut bs = bs;
172
173        // Fast path: single contiguous chunk.
174        // If chunk length equals remaining bytes, the entire buffer is contiguous.
175        // This avoids Vec allocation and loop overhead for the common case.
176        if bs.chunk().len() == bs.remaining() {
177            let bytes = bs.copy_to_bytes(bs.remaining());
178            return self.write(Buffer::from(bytes)).await;
179        }
180
181        // Slow path: multiple chunks (e.g., Chain<T, U>).
182        // Iterate over each chunk and collect them into a Vec<Bytes>.
183        // For Bytes chunks, copy_to_bytes is zero-copy.
184        // For other types, only the current chunk is copied, not the entire buffer.
185        let mut chunks = Vec::new();
186        while bs.has_remaining() {
187            let chunk_len = bs.chunk().len();
188            chunks.push(bs.copy_to_bytes(chunk_len));
189        }
190        self.write(Buffer::from(chunks)).await
191    }

193    /// Copy a range of an existing object into this writer.
194    ///
195    /// The source path is resolved by the same [`Operator`] that created this
196    /// writer. Calls to `write` and `copy_from` may be interleaved, and their
197    /// call order defines the destination bytes. OpenDAL uses a native
198    /// writer-side copy when the composed service supports it and otherwise
199    /// streams the range through the client. Check
200    /// [`Capability::write_can_copy_from`] to determine whether eligible
201    /// non-append writes use the native path.
202    ///
203    /// The source path must differ from the writer's destination path.
204    /// Copying a file onto itself returns [`ErrorKind::IsSameFile`].
205    ///
206    /// Each call accepts at most 5 GiB. Applications must split larger source
207    /// ranges into multiple ordered calls.
208    ///
209    /// If a call fails after the underlying writer accepts part of the range,
210    /// retrying the same path, range, and read options resumes after the
211    /// accepted prefix instead of copying it again. Until that retry succeeds
212    /// or [`Writer::abort`] runs, other writes and [`Writer::close`] fail.
213    pub async fn copy_from(&mut self, path: &str, range: impl Into<BytesRange>) -> Result<()> {
214        self.copy_from_options(
215            path,
216            options::ReadOptions {
217                range: range.into(),
218                ..Default::default()
219            },
220        )
221        .await
222    }
223
224    /// Copy an existing object range with read conditions and execution options.
225    ///
226    /// [`options::ReadOptions::limit`] bounds the number of bytes copied from
227    /// the start of the selected range and accepts EOF after a valid starting
228    /// byte, matching [`Operator::read_options`]. A zero limit copies nothing.
229    /// A non-zero limit on a forward selection that starts at or beyond EOF
230    /// returns [`crate::ErrorKind::RangeNotSatisfied`].
231    /// Combining a limit with a positive `chunk` or with `concurrent` greater
232    /// than one returns [`ErrorKind::ConfigInvalid`].
233    ///
234    /// The source path must differ from the writer's destination path unless
235    /// [`options::ReadOptions::version`] selects a specific source version to
236    /// restore. Copying the current object onto itself returns
237    /// [`ErrorKind::IsSameFile`].
238    ///
239    /// A retry with the same path, range, and read options resumes a partially
240    /// accepted copy instead of replaying its accepted prefix. Any other write
241    /// or [`Writer::close`] fails until the retry succeeds or [`Writer::abort`]
242    /// runs.
243    pub async fn copy_from_options(
244        &mut self,
245        path: &str,
246        options: options::ReadOptions,
247    ) -> Result<()> {
248        let path = normalize_path(path);
249        if !validate_path(&path, EntryMode::FILE) {
250            return Err(
251                Error::new(ErrorKind::IsADirectory, "copy source path is a directory")
252                    .with_operation("Writer::copy_from")
253                    .with_context("service", self.ctx.service().info().scheme())
254                    .with_context("path", &path),
255            );
256        }
257        if path == self.ctx.path() && options.version.is_none() {
258            return Err(Error::new(
259                ErrorKind::IsSameFile,
260                "source and destination paths are the same",
261            )
262            .with_operation("Writer::copy_from")
263            .with_context("service", self.ctx.service().info().scheme())
264            .with_context("source", &path)
265            .with_context("destination", self.ctx.path()));
266        }
267        if options.limit.is_some()
268            && (options.chunk.is_some_and(|chunk| chunk != 0) || options.concurrent > 1)
269        {
270            return Err(Error::new(
271                ErrorKind::ConfigInvalid,
272                "read limit cannot be combined with chunk or concurrent reads",
273            ));
274        }
275        let limit = options.limit;
276        let (range, args, options) = options.into();
277        self.inner
278            .copy_from(&path, args, options, range, limit)
279            .await
280    }

302    /// Convert this writer into a [`BufferSink`] that implements
303    /// [`Sink<Buffer>`][futures::Sink].
304    ///
305    /// # Notes
306    ///
307    /// BufferSink is a zero-cost abstraction. The underlying writer
308    /// will reuse the Bytes and won't perform any copy operation over data.
309    ///
310    /// # Examples
311    ///
312    /// ## Basic Usage
313    ///
314    /// ```
315    /// use std::io;
316    ///
317    /// use bytes::Bytes;
318    /// use futures::SinkExt;
319    /// use opendal_core::Buffer;
320    /// use opendal_core::Operator;
321    /// use opendal_core::Result;
322    ///
323    /// async fn test(op: Operator) -> io::Result<()> {
324    ///     let mut s = op.writer("hello.txt").await?.into_sink();
325    ///     let bs = "Hello, World!".as_bytes();
326    ///     s.send(Buffer::from(bs)).await?;
327    ///     s.close().await?;
328    ///
329    ///     Ok(())
330    /// }
331    /// ```
332    ///
333    /// ## Concurrent Write
334    ///
335    /// ```
336    /// use std::io;
337    ///
338    /// use bytes::Bytes;
339    /// use futures::SinkExt;
340    /// use opendal_core::Buffer;
341    /// use opendal_core::Operator;
342    /// use opendal_core::Result;
343    ///
344    /// async fn test(op: Operator) -> io::Result<()> {
345    ///     let mut w = op
346    ///         .writer_with("hello.txt")
347    ///         .concurrent(8)
348    ///         .chunk(256)
349    ///         .await?
350    ///         .into_sink();
351    ///     let bs = "Hello, World!".as_bytes();
352    ///     w.send(Buffer::from(bs)).await?;
353    ///     w.close().await?;
354    ///
355    ///     Ok(())
356    /// }
357    /// ```
358    pub fn into_sink(self) -> BufferSink {
359        BufferSink::new(self.inner)
360    }

547    /// `copy_from_options` must copy the same bytes that `read_options`
548    /// returns for the same options.
549    #[tokio::test]
550    async fn test_writer_copy_from_options_honors_limit() -> Result<()> {
551        let op = Operator::new(services::Memory::default())?;
552        op.write("source", "0123456789").await?;
553
554        let cases = [
555            // A zero limit copies nothing.
556            (
557                options::ReadOptions {
558                    range: (0..6).into(),
559                    limit: Some(0),
560                    ..Default::default()
561                },
562                b"".as_slice(),
563            ),
564            // The limit bounds the supplied range from its start.
565            (
566                options::ReadOptions {
567                    range: (0..6).into(),
568                    limit: Some(3),
569                    ..Default::default()
570                },
571                b"012".as_slice(),
572            ),
573            // A limit larger than the supplied range keeps the whole range.
574            (
575                options::ReadOptions {
576                    range: (0..6).into(),
577                    limit: Some(10),
578                    ..Default::default()
579                },
580                b"012345".as_slice(),
581            ),
582            // The limit applies to an open-ended range.
583            (
584                options::ReadOptions {
585                    range: (4_u64..).into(),
586                    limit: Some(3),
587                    ..Default::default()
588                },
589                b"456".as_slice(),
590            ),
591            // The limit applies to a suffix range.
592            (
593                options::ReadOptions {
594                    range: BytesRange::suffix(4),
595                    limit: Some(2),
596                    ..Default::default()
597                },
598                b"67".as_slice(),
599            ),
600            // A limit accepts EOF after a valid starting byte.
601            (
602                options::ReadOptions {
603                    range: (0..12).into(),
604                    limit: Some(20),
605                    ..Default::default()
606                },
607                b"0123456789".as_slice(),
608            ),
609        ];
610
611        for (idx, (options, expected)) in cases.into_iter().enumerate() {
612            assert_eq!(
613                op.read_options("source", options.clone()).await?.to_vec(),
614                expected,
615                "read_options must return the expected bytes for case {idx}"
616            );
617
618            let target = format!("target-{idx}");
619            let mut writer = op.writer(&target).await?;
620            writer.copy_from_options("source", options).await?;
621            writer.close().await?;
622            assert_eq!(
623                op.read(&target).await?.to_vec(),
624                expected,
625                "copy_from_options must copy the same bytes for case {idx}"
626            );
627        }
628        Ok(())
629    }

690        let mut chain_same = part1.clone().chain(part2.clone());
691        let chain = part1.chain(part2);
692
693        let mut writer = op.writer(path).await.unwrap();
694        writer.write_from(chain).await.expect("write must succeed");
695        writer.close().await.expect("close must succeed");
696
697        let buf = op.read(path).await.expect("read to end mut succeed");
698
699        assert_eq!(
700            buf.to_bytes(),
701            chain_same.copy_to_bytes(chain_same.remaining())
702        );
703    }
