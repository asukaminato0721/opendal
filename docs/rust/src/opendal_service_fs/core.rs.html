38    pub fn root_join(&self, path: &str) -> Result<PathBuf> {
39        confined_join(&self.root, path)
40    }

42    // Build write path and ensure the parent dirs created
43    pub async fn ensure_write_abs_path(&self, parent: &Path, path: &str) -> Result<PathBuf> {
44        self.ensure_write_abs_path_with(parent, path, new_std_io_error)
45            .await
46    }
47
48    async fn ensure_write_abs_path_with(
49        &self,
50        parent: &Path,
51        path: &str,
52        map_io_error: impl FnOnce(std::io::Error) -> Error,
53    ) -> Result<PathBuf> {
54        let p = confined_join(parent, path)?;
55
56        // Create dir before write path.
57        //
58        // TODO(xuanwo): There are many works to do here:
59        //   - Is it safe to create dir concurrently?
60        //   - Do we need to extract this logic as new util functions?
61        //   - Is it better to check the parent dir exists before call mkdir?
62        let parent = PathBuf::from(&p)
63            .parent()
64            .ok_or_else(|| {
65                Error::new(
66                    ErrorKind::Unexpected,
67                    "path should have parent but not, it must be malformed",
68                )
69                .with_context("input", p.to_string_lossy())
70            })?
71            .to_path_buf();
72
73        tokio::fs::create_dir_all(&parent)
74            .await
75            .map_err(map_io_error)?;
76
77        Ok(p)
78    }

80    pub async fn fs_create_dir(&self, path: &str) -> Result<()> {
81        let p = self.root_join(path)?;
82        tokio::fs::create_dir_all(&p)
83            .await
84            .map_err(new_std_io_error)?;
85        Ok(())
86    }

88    pub async fn fs_stat(&self, path: &str) -> Result<Metadata> {
89        let p = self.root_join(path)?;
90        let meta = tokio::fs::metadata(&p).await.map_err(new_std_io_error)?;
91        let mode = if meta.is_dir() {
92            EntryMode::DIR
93        } else if meta.is_file() {
94            EntryMode::FILE
95        } else {
96            EntryMode::Unknown
97        };
98        let mut m = match mode {
99            EntryMode::FILE => MetadataBuilder::file(meta.len()),
100            EntryMode::DIR => MetadataBuilder::dir(),
101            EntryMode::Unknown => MetadataBuilder::unknown(),
102        };
103        m.last_modified(Timestamp::try_from(
104            meta.modified().map_err(new_std_io_error)?,
105        )?);
106
107        // Read user metadata from xattr on Unix systems
108        #[cfg(unix)]
109        {
110            let user_metadata = Self::get_user_metadata(&p)?;
111            if !user_metadata.is_empty() {
112                m.user_metadata(user_metadata);
113            }
114        }
115
116        Ok(m.build())
117    }

119    pub async fn fs_list(&self, path: &str) -> Result<Option<tokio::fs::ReadDir>> {
120        let p = self.root_join(path)?;
121
122        match tokio::fs::read_dir(&p).await {
123            Ok(rd) => Ok(Some(rd)),
124            Err(e) => {
125                match e.kind() {
126                    // Return empty list if the directory not found
127                    std::io::ErrorKind::NotFound => Ok(None),
128                    // If the path is not a directory, return an empty list
129                    //
130                    // The path could be a file or a symbolic link in this case.
131                    // Returning a NotADirectory error to the user isn't helpful; instead,
132                    // providing an empty directory is a more user-friendly. In fact, the dir
133                    // `path/` does not exist.
134                    std::io::ErrorKind::NotADirectory => Ok(None),
135                    _ => Err(new_std_io_error(e)),
136                }
137            }
138        }
139    }

141    pub async fn fs_copy(&self, from: &str, to: &str) -> Result<u64> {
142        let from = self.root_join(from)?;
143        // try to get the metadata of the source file to ensure it exists
144        tokio::fs::metadata(&from).await.map_err(new_std_io_error)?;
145
146        // Read source user metadata before copying: copying onto the same path
147        // would otherwise read the replaced destination. A source whose user
148        // metadata cannot be read keeps the previous copy behavior of not
149        // failing the content copy.
150        #[cfg(unix)]
151        let user_metadata = Self::get_user_metadata(&from).ok();
152
153        let to = self
154            .ensure_write_abs_path(&self.root, to.trim_end_matches('/'))
155            .await?;
156
157        let size = tokio::fs::copy(&from, &to)
158            .await
159            .map_err(new_std_io_error)?;
160
161        // Copying truncates an existing destination in place, so stale user
162        // metadata from the previous file must be replaced explicitly.
163        #[cfg(unix)]
164        if let Some(user_metadata) = user_metadata {
165            Self::replace_user_metadata(&to, Some(&user_metadata))?;
166        }
167
168        Ok(size)
169    }

171    pub async fn fs_rename(&self, from: &str, to: &str, if_not_exists: bool) -> Result<()> {
172        #[cfg(not(target_os = "linux"))]
173        if if_not_exists {
174            return Err(Error::new(
175                ErrorKind::Unsupported,
176                "Conditional rename is only supported on Linux",
177            ));
178        }
179
180        let map_path_error = |err| {
181            #[cfg(target_os = "linux")]
182            if if_not_exists {
183                return parse_conditional_rename_directory_error(err, new_std_io_error);
184            }
185            new_std_io_error(err)
186        };
187
188        let from = self.root_join(from)?;
189        tokio::fs::metadata(&from).await.map_err(map_path_error)?;
190
191        let to = self
192            .ensure_write_abs_path_with(&self.root, to.trim_end_matches('/'), map_path_error)
193            .await?;
194
195        #[cfg(target_os = "linux")]
196        if if_not_exists {
197            return tokio::task::spawn_blocking(move || rename_noreplace(&from, &to))
198                .await
199                .map_err(new_task_join_error)?;
200        }
201
202        tokio::fs::rename(from, to)
203            .await
204            .map_err(new_std_io_error)?;
205        Ok(())
206    }

208    /// Get user metadata from extended attributes on Unix systems.
209    ///
210    /// Reads xattr in the "user." namespace and strips the prefix, allowing
211    /// interoperability with other applications that set user xattr.
212    #[cfg(unix)]
213    pub fn get_user_metadata(path: &Path) -> Result<HashMap<String, String>> {
214        let mut user_metadata = HashMap::new();
215
216        let attrs = match xattr::list(path) {
217            Ok(attrs) => attrs,
218            Err(e) if is_xattr_unsupported(&e) => return Ok(user_metadata),
219            Err(e) => return Err(new_std_io_error(e)),
220        };
221
222        for attr in attrs {
223            let attr_name = attr.to_string_lossy();
224            // Only read xattr in the "user." namespace and strip the prefix
225            if let Some(key) = attr_name.strip_prefix(XATTR_USER_PREFIX)
226                && let Ok(Some(value)) = xattr::get(path, &attr)
227                && let Ok(v) = String::from_utf8(value)
228            {
229                user_metadata.insert(key.to_string(), v);
230            }
231        }
232
233        Ok(user_metadata)
234    }
235
236    /// Set user metadata as extended attributes on Unix systems.
237    ///
238    /// Keys are automatically prefixed with "user." to comply with Linux xattr
239    /// naming requirements. This allows interoperability with other applications.
240    #[cfg(unix)]
241    pub fn set_user_metadata(path: &Path, user_metadata: &HashMap<String, String>) -> Result<()> {
242        for (key, value) in user_metadata {
243            let attr_name = format!("{XATTR_USER_PREFIX}{key}");
244            xattr::set(path, &attr_name, value.as_bytes()).map_err(new_std_io_error)?;
245        }
246        Ok(())
247    }
248
249    /// Replace the user metadata on `path` with `user_metadata`.
250    ///
251    /// Only extended attributes in the "user." namespace are removed; other
252    /// namespaces are never touched. Passing `None` clears all user metadata.
253    /// Filesystems without xattr support are treated as having no user metadata
254    /// so that ordinary writes and copies do not fail there.
255    #[cfg(unix)]
256    pub fn replace_user_metadata(
257        path: &Path,
258        user_metadata: Option<&HashMap<String, String>>,
259    ) -> Result<()> {
260        // Write the new values first so a failure while setting metadata keeps
261        // the previous user metadata instead of clearing it first.
262        if let Some(user_metadata) = user_metadata {
263            Self::set_user_metadata(path, user_metadata)?;
264        }
265
266        let attrs = match xattr::list(path) {
267            Ok(attrs) => attrs,
268            Err(e) if is_xattr_unsupported(&e) => return Ok(()),
269            Err(e) => return Err(new_std_io_error(e)),
270        };
271
272        for attr in attrs {
273            let attr_name = attr.to_string_lossy();
274            let Some(key) = attr_name.strip_prefix(XATTR_USER_PREFIX) else {
275                continue;
276            };
277            if user_metadata.is_some_and(|metadata| metadata.contains_key(key)) {
278                continue;
279            }
280
281            match xattr::remove(path, &attr) {
282                Ok(()) => {}
283                // Another task may have removed the attribute already.
284                Err(e) if is_xattr_not_found(&e) => {}
285                Err(e) if is_xattr_unsupported(&e) => return Ok(()),
286                Err(e) => return Err(new_std_io_error(e)),
287            }
288        }
289
290        Ok(())
291    }
292}

294/// Prefix for user xattr namespace on Linux.
295/// Using "user." as the standard namespace for user-defined attributes.
296#[cfg(unix)]
297const XATTR_USER_PREFIX: &str = "user.";
298
299/// Whether an xattr error means the filesystem does not support xattrs.
300#[cfg(unix)]
301fn is_xattr_unsupported(err: &std::io::Error) -> bool {
302    if err.kind() == std::io::ErrorKind::Unsupported {
303        return true;
304    }
305
306    // Some platforms leave ENOTSUP/EOPNOTSUPP/ENOSYS uncategorized.
307    #[cfg(target_os = "linux")]
308    {
309        use rustix::io::Errno;
310        matches!(
311            err.raw_os_error().map(Errno::from_raw_os_error),
312            Some(Errno::OPNOTSUPP | Errno::NOSYS)
313        )
314    }
315
316    #[cfg(not(target_os = "linux"))]
317    {
318        false
319    }
320}
321
322/// Whether an xattr error means the attribute is already gone.
323///
324/// `xattr` reports a missing attribute as `ENODATA` on Linux and `ENOATTR` on
325/// macOS and the BSDs. Neither is mapped to [`std::io::ErrorKind::NotFound`].
326/// `ENOENT` reports a missing file, so it must keep surfacing as an error.
327#[cfg(unix)]
328fn is_xattr_not_found(err: &std::io::Error) -> bool {
329    let Some(code) = err.raw_os_error() else {
330        return false;
331    };
332
333    // Platforms that `xattr` does not support report `Unsupported` for every
334    // call, so only the supported ones need a missing-attribute errno.
335    #[cfg(any(target_os = "linux", target_os = "android", target_os = "hurd"))]
336    {
337        code == libc::ENODATA
338    }
339
340    #[cfg(any(target_os = "macos", target_os = "freebsd", target_os = "netbsd"))]
341    {
342        code == libc::ENOATTR
343    }
344
345    #[cfg(not(any(
346        target_os = "linux",
347        target_os = "android",
348        target_os = "hurd",
349        target_os = "macos",
350        target_os = "freebsd",
351        target_os = "netbsd"
352    )))]
353    {
354        let _ = code;
355        false
356    }
357}

359/// Publish a staged write at `to` without replacing an existing entry.
360///
361/// Returns `ConditionNotMatch` when the destination already exists.
362pub(super) fn publish_staged_file(from: &Path, to: &Path) -> Result<()> {
363    #[cfg(target_os = "linux")]
364    {
365        rename_noreplace(from, to)
366    }
367
368    #[cfg(not(target_os = "linux"))]
369    {
370        // Non-Linux systems lack a rename that fails on an existing
371        // destination. Linking the staged file to `to` creates the destination
372        // atomically and fails with `AlreadyExists` when it is already
373        // present, on both POSIX and Windows. Hard links require both paths on
374        // one filesystem, which the atomic publish already requires for its
375        // rename.
376        std::fs::hard_link(from, to)
377            .map_err(|err| parse_error(ErrorContext::new(ServiceOperation("Rename")), err))?;
378
379        match std::fs::remove_file(from) {
380            // A concurrent cleanup or an unusual filesystem may have removed
381            // the staging entry already; the destination is still published.
382            Err(err) if err.kind() == std::io::ErrorKind::NotFound => Ok(()),
383            // The destination was published, but the staging file is orphaned
384            // now. Report it instead of leaking it silently.
385            Err(err) => {
386                Err(new_std_io_error(err).with_context("service_operation", "RemoveStagedFile"))
387            }
388            Ok(()) => Ok(()),
389        }
390    }
391}

393#[cfg(target_os = "linux")]
394pub(super) fn rename_noreplace(from: &Path, to: &Path) -> Result<()> {
395    use rustix::fs::{CWD, RenameFlags, renameat_with};
396    use rustix::io::Errno;
397
398    renameat_with(CWD, from, CWD, to, RenameFlags::NOREPLACE).map_err(|err| match err {
399        Errno::INVAL | Errno::NOSYS | Errno::OPNOTSUPP => Error::new(
400            ErrorKind::Unsupported,
401            "The kernel or filesystem does not support conditional rename",
402        )
403        .set_source(err),
404        _ => parse_conditional_rename_directory_error(err.into(), |err| {
405            parse_error(ErrorContext::new(ServiceOperation("Rename")), err)
406        }),
407    })
408}

410#[cfg(target_os = "linux")]
411fn parse_conditional_rename_directory_error(
412    err: std::io::Error,
413    fallback: impl FnOnce(std::io::Error) -> Error,
414) -> Error {
415    use rustix::io::Errno;
416
417    let kind = match err.raw_os_error().map(Errno::from_raw_os_error) {
418        Some(Errno::ISDIR) => ErrorKind::IsADirectory,
419        Some(Errno::NOTDIR) => ErrorKind::NotADirectory,
420        _ => return fallback(err),
421    };
422
423    Error::new(kind, err.kind().to_string())
424        .set_source(err)
425        .with_context("service_operation", "Rename")
426}

428/// Context needed to classify an error from this service.
429#[derive(Clone, Copy, Debug)]
430pub(crate) struct ErrorContext {
431    service_operation: ServiceOperation,
432}
433
434impl ErrorContext {
435    pub(crate) const fn new(service_operation: ServiceOperation) -> Self {
436        Self { service_operation }
437    }
438}

440/// Parse an error using its service operation context.
441pub(crate) fn parse_error(ctx: ErrorContext, err: std::io::Error) -> Error {
442    let err = match err.kind() {
443        std::io::ErrorKind::AlreadyExists => Error::new(
444            ErrorKind::ConditionNotMatch,
445            "The file already exists in the filesystem",
446        )
447        .set_source(err),
448        _ => new_std_io_error(err),
449    };
450
451    err.with_context("service_operation", ctx.service_operation.0)
452}
